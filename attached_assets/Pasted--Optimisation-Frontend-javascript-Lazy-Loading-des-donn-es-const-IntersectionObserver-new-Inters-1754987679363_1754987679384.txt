function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

class FraudDetectionModel:
    def __init__(self):
        self.model = IsolationForest(contamination=0.1, random_state=42)
        self.scaler = StandardScaler()
        self.features = ['amount', 'days_to_due', 'client_risk_score', 
                        'hour_of_day', 'day_of_week', 'depositor_frequency']
    
    def prepare_features(self, cheque):
        return {
            'amount': cheque.amount,
            'days_to_due': (cheque.due_date - cheque.issue_date).days,
            'client_risk_score': cheque.client.risk_score,
            'hour_of_day': cheque.created_at.hour,
            'day_of_week': cheque.created_at.weekday(),
            'depositor_frequency': self.get_depositor_frequency(cheque.depositor_id)
        }
    
    def predict_fraud_probability(self, cheque):
        features = self.prepare_features(cheque)
        feature_array = self.scaler.transform([list(features.values())])
        anomaly_score = self.model.decision_function(feature_array)[0]
        return max(0, min(1, (anomaly_score + 0.5) * 2))

pythonimport hashlib
import json

class BlockchainAudit:
    def __init__(self):
        self.chain = []
        self.create_genesis_block()
    
    def create_genesis_block(self):
        genesis_block = {
            'index': 0,
            'timestamp': datetime.utcnow().isoformat(),
            'data': 'Genesis Block',
            'previous_hash': '0',
            'hash': self.calculate_hash(0, datetime.utcnow().isoformat(), 'Genesis Block', '0')
        }
        self.chain.append(genesis_block)
    
    def add_transaction_block(self, transaction_data):
        previous_block = self.chain[-1]
        new_block = {
            'index': len(self.chain),
            'timestamp': datetime.utcnow().isoformat(),
            'data': transaction_data,
            'previous_hash': previous_block['hash'],
            'hash': None
        }
        new_block['hash'] = self.calculate_hash(
            new_block['index'],
            new_block['timestamp'],
            new_block['data'],
            new_block['previous_hash']
        )
        self.chain.append(new_block)
        return new_block
    
    def calculate_hash(self, index, timestamp, data, previous_hash):
        block_string = f"{index}{timestamp}{json.dumps(data, sort_keys=True)}{previous_hash}"
        return hashlib.sha256(block_string.encode()).hexdigest()

pythonfrom transformers import GPT2LMHeadModel, GPT2Tokenizer

class ReportGenerator:
    def __init__(self):
        self.model = GPT2LMHeadModel.from_pretrained('gpt2')
        self.tokenizer = GPT2Tokenizer.from_pretrained('gpt2')
    
    def generate_executive_summary(self, data):
        prompt = f"""
        Résumé exécutif - Système de Gestion des Chèques
        
        Données de la période:
        - Total chèques traités: {data['total_cheques']}
        - Montant total: {data['total_amount']:,.2f} DH
        - Taux d'encaissement: {data['success_rate']:.1%}
        - Taux de rejet: {data['rejection_rate']:.1%}
        
        Analyse:
        """
        
        inputs = self.tokenizer.encode(prompt, return_tensors='pt')
        outputs = self.model.generate(
            inputs,
            max_length=inputs.shape[1] + 200,
            num_return_sequences=1,
            temperature=0.7,
            pad_token_id=self.tokenizer.eos_token_id
        )
        
        generated_text = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
        return generated_text[len(prompt):]
